fast BOUND:
  def __init__(self,
                mu= 1**2,                 # intrinsic amplitude, converges to sqrt(mu)
                omega_swing = 8*2*np.pi,   # frequency in swing phase (can edit)
                omega_stance = 9*2*np.pi,  # frequency in stance phase (can edit)
                gait="BOUND",             # Gait, can be TROT, WALK, PACE, BOUND, etc.
                alpha=50,                # amplitude convergence factor
                coupling_strength=3,     # coefficient to multiply coupling matrix
                couple=True,             # whether oscillators should be coupled
                time_step=0.001,         # time step 
                ground_clearance=0.07,   # foot swing height                         #INITIALLY 0.07
                ground_penetration=0.01, # foot stance penetration into ground 
                robot_height=0.3,        # in nominal case (standing) 
                des_step_len=0.1,       # desired step length                      #INITIALLY 0.05
                max_step_len_rl=0.13,     # max step length, for RL scaling 
                use_RL=False             # whether to learn parameters with RL 
                ):

kp=np.array([200, 40, 350])
kd=np.array([1,6,8])
# Cartesian PD gains
kpCartesian = np.diag([500, 350, 350])
kdCartesian = np.diag([20]*3)
